import datetime, json, os, pathlib, resource, subprocess, sys, time
root=pathlib.Path('/workspace/bagelito-validation')
out=root/'docs/qa/product-checkout-independent-20261102'
label=sys.argv[1]; command=sys.argv[2:]
env={'PATH':os.environ['PATH'],'LANG':'C.UTF-8','NODE_OPTIONS':'--max-old-space-size=512 --max-semi-space-size=4','CIRCLE_NODE_TOTAL':'1','RAYON_NUM_THREADS':'1','UV_THREADPOOL_SIZE':'1'}
start=time.monotonic(); timestamp=datetime.datetime.now(datetime.timezone.utc).isoformat()
with (out/(label+'.log')).open('w') as log:
    log.write('Command: '+json.dumps(command)+'\n'); log.flush()
    result=subprocess.run(command,cwd=root,env=env,stdout=log,stderr=subprocess.STDOUT)
record={'label':label,'command':command,'started_utc':timestamp,'elapsed_seconds':round(time.monotonic()-start,3),'returncode':result.returncode,'environment':'minimal explicit allowlist; no inherited credentials','node_options':env['NODE_OPTIONS'],'max_child_rss_kib':resource.getrusage(resource.RUSAGE_CHILDREN).ru_maxrss}
(out/(label+'.json')).write_text(json.dumps(record,indent=2)+'\n')
print(json.dumps(record)); print((out/(label+'.log')).read_text()[-4000:])
sys.exit(result.returncode)
